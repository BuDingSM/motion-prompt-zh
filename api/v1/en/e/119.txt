// Light leak / ライトリーク
// Pack: photo (Photo looks, lighting & styles) · element 23 · Category: Tones & finishing looks
// What happens: Red, orange and yellow light bleeds into the beach photo from the right edge and bottom left, slowly changing shape.
// Motion spec:
//   - Light: 4 elliptical radial gradients — red rgb(255,60,20) at the right edge (280×140 px), orange rgb(255,150,30) lower right, yellow rgb(255,210,90) at the top, red-orange rgb(255,90,40) bottom left
//   - Blending: hard light to tint, then screen at 80% to bloom; a 140 px orange-white band at the right edge (50%)
//   - Motion: each drifts 14–24 px back and forth (one swing per 5 s)
//   - A slider at bottom right shows the amount
//   - Timing: the effect comes in from 0.6 s to 1.7 s, holds until 3.9 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// When to use it: For nostalgic, summery, accidental-looking memories.
// Tags: light leak, film leak, vintage, analog, lomo, warm glow
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#119
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 23
// 23 ライトリーク: フィルムに光がもれたように、画面の端から赤・橙・黄の光がにじみ込み、ゆっくり形を変える
CELLS[22] = cell((k, w, h, u) => {
  const a = env(u), t = u / CYC * TAU;
  k.drawImage(beach(), 0, 0);
  if (a > 0.001) {
    const blob = (x, y, rx, ry, c, al) => {
      k.save(); k.translate(x, y); k.scale(1, ry / rx);
      const g = k.createRadialGradient(0, 0, 0, 0, 0, rx); g.addColorStop(0, rgb(c, al * a)); g.addColorStop(0.5, rgb(c, al * a * 0.55)); g.addColorStop(1, rgb(c, 0));
      k.fillStyle = g; k.fillRect(-rx, -rx, rx * 2, rx * 2); k.restore();
    };
    const leaks = () => {
      blob(CW + 10, 50 + 24 * Math.sin(t), 280, 140, [255, 60, 20], 1);
      blob(CW - 50, 140 + 18 * Math.cos(t), 200, 90, [255, 150, 30], 0.95);
      blob(CW - 170 + 20 * Math.sin(t + 1), 16, 140, 56, [255, 210, 90], 0.75);
      blob(-30, 150 + 14 * Math.sin(t + 2), 170, 110, [255, 90, 40], 0.85);
    };
    k.globalCompositeOperation = 'hard-light'; leaks();   // 色をのせる
    k.globalCompositeOperation = 'screen'; k.globalAlpha = 0.8; leaks(); k.globalAlpha = 1;   // 明るくにじませる
    const band = k.createLinearGradient(CW - 140, 0, CW, 0); band.addColorStop(0, 'rgba(255,120,60,0)'); band.addColorStop(1, rgb([255, 200, 150], 0.5 * a));
    k.fillStyle = band; k.fillRect(0, 0, CW, CH);
    k.globalCompositeOperation = 'source-over';
  }
  slider(k, a);
});
