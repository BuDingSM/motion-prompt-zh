// フィルムグレイン / Film grain
// パック: photo (写真の表現・光と画風) · element 22 · カテゴリ: 仕上げ・画風
// 何が起きるか: 海辺の写真の全体に細かい粒子が乗り、黒が少し浮いて暖かい色になる。粒の模様は少しずつ入れかわる。
// 動きの仕様:
//   - 粒子: 画素ごとの灰色のノイズ（一様乱数3つの和、±96）を0.45pxぼかし、オーバーレイで70%
//   - 動き: 固定シードの4枚の模様を1秒に8回順に入れかえる（5秒で40回）
//   - 色: R = 22 + 0.88R、G = 16 + 0.86G、B = 10 + 0.8B（黒が浮いて暖かくなる）
//   - 右下のスライダーが効き具合
//   - 時間: 0.6〜1.7秒で効果が入り、3.9秒まで保ち、4.8秒までに元へ戻る（1周5秒、3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当））
// 使いどころ: デジタルのつるっとした感じを消して、フィルムの手触りを出したいとき。
// タグ: film grain, grain, noise, analog, 35mm film, texture, vintage
// サイズ: 539x208 px · 1周: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#118
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 22
// 22 フィルムグレイン: 細かい粒子が画面全体に乗り、黒が少し浮いて暖かい色になる。粒は1秒に8回、4枚の模様を順に入れかえる
function grainTiles() {
  return once('grain', () => {
    const r = rng(22), out = [];
    for (let n = 0; n < 4; n++) {
      const c = mkCanvas(CW, CH), k = c.getContext('2d'), img = k.createImageData(CW, CH), d = img.data;
      for (let i = 0; i < d.length; i += 4) { const v = 128 + (r() + r() + r() - 1.5) * 64; d[i] = d[i + 1] = d[i + 2] = v; d[i + 3] = 255; }
      k.putImageData(img, 0, 0);
      const c2 = mkCanvas(CW, CH), k2 = c2.getContext('2d'); k2.filter = 'blur(0.45px)'; k2.drawImage(c, 0, 0); out.push(c2);
    }
    return out;
  });
}
CELLS[21] = cell((k, w, h, u) => {
  const a = env(u);
  k.drawImage(beach(), 0, 0);
  if (a > 0.001) {
    k.globalAlpha = a; k.drawImage(graded('film', p => [22 + p[0] * 0.88, 16 + p[1] * 0.86, 10 + p[2] * 0.8]), 0, 0);
    k.globalCompositeOperation = 'overlay'; k.globalAlpha = 0.7 * a; k.drawImage(grainTiles()[Math.floor(u * 8) % 4], 0, 0);
    k.globalCompositeOperation = 'source-over'; k.globalAlpha = 1;
  }
  slider(k, a);
});
