// 動態殘影 / Motion trail
// 類別包: motion (動態圖形動效) · element 14 · 分類: 動態質感
// 效果說明: A ball moving in a figure eight leaves a fading trail of copies behind it.
// 動效規格:
//   - A ball (coral, radius 25, white centre radius 11) traces a figure eight once per 4.7 s: x = 196·sin(a), y = 62·sin(2a), with a advancing at constant speed.
//   - 15 copies, each placed 0.105 rad (about 0.079 s) further back along the path.
//   - Copy radius goes 22→6 and opacity is 0.9 × (1 − i/15) + 0.05, so older copies are smaller and fainter.
//   - Copy colour runs coral → yellow → blue from newest to oldest.
//   - Draw the oldest copy first and the ball itself last.
// 適用情境: Use it to show speed or a path: flying objects, cursors, points of light.
// 標籤: trail, motion-trail, echo, afterimage, ghost, path, speed
// 尺寸: 539x208 px · 週期: 4.7 s
// 依賴: kit motion -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/motion.js
// 動態範例: https://budingsm.github.io/motion-prompt-zh/#014
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫方已清空畫布並完成 save/restore），w、h = 尺寸，u = [0, cycle) 內的秒數。畫面只由 u 決定。
// 以下是元素程式碼。CELLS、常數和輔助函式位於 kit 中。

// @element 14
// 14 モーショントレイル（動いた跡に、残像が尾を引く）
CELLS[13] = (k, w, h, u) => {
  stage(k);
  const X = w / 2, Y = h / 2 + 2, A = 196, B = 62, th = u / CYC * Math.PI * 2;
  const pt = a => [X + A * Math.sin(a), Y + B * Math.sin(2 * a)];
  const N = 15;
  for (let i = N; i >= 1; i--) {
    const [x, y] = pt(th - i * 0.105), f = i / N;
    k.globalAlpha = 0.9 * (1 - f) + 0.05; k.fillStyle = f < 0.5 ? mix(C.coral, C.yel, f * 2) : mix(C.yel, C.blue, (f - 0.5) * 2);
    k.beginPath(); k.arc(x, y, lerp(22, 6, f), 0, 7); k.fill();
  }
  k.globalAlpha = 1; const [x, y] = pt(th);
  k.fillStyle = C.coral; k.beginPath(); k.arc(x, y, 25, 0, 7); k.fill();
  k.fillStyle = '#fff'; k.beginPath(); k.arc(x, y, 11, 0, 7); k.fill();
};
