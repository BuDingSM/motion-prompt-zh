// Badge / Badge
// 類別包: ui (UI 組件名稱) · element 23 · 分類: 導覽與狀態元件
// 效果說明: Each time a notification arrives, the blue dot on the bell in the header counts up 1, 2, 3; clicking the bell clears it.
// 動效規格:
//   - 4.5 s cycle on a 542×191 card. A 54-tall header with a line-drawn bell at (482, 27) on the right.
//   - Badge: a blue circle of radius 12 at the bell's top right (+12, −12), 2.5 px white rim, white 14 px number.
//   - Notifications arrive at 0.7, 1.3 and 1.9 s: the count goes 1 → 2 → 3. The first scales in over 0.7–0.9 s with easeOutCubic.
//   - Each arrival pops the scale from 1.3 back to 1 over 0.3 s and sends out a ripple ring (radius 12 → 28, opacity 0.5 → 0) over 0.5 s.
//   - The cursor reaches the bell at 2.95 s and clicks at 3.0 s; the badge shrinks to 0 over 3.0–3.25 s with easeInOutCubic.
// 適用情境: Use it to call attention to a count, such as unread messages or items in a cart.
// 標籤: badge, notification-badge, counter, count, dot, unread
// 尺寸: 542x191 px · 週期: 4.5 s
// 依賴: kit ui -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/ui.js
// 動態範例: https://budingsm.github.io/motion-prompt-zh/#071
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫方已清空畫布並完成 save/restore），w、h = 尺寸，u = [0, cycle) 內的秒數。畫面只由 u 決定。
// 以下是元素程式碼。CELLS、常數和輔助函式位於 kit 中。

// @element 23
// 23 バッジ（Badge）
CELLS[22] = ui((w, h, t) => {
  X.fillStyle = '#fff'; X.fillRect(0, 0, w, 54); X.fillStyle = C.line; X.fillRect(0, 54, w, 1.5);
  fillRR(24, 19, 76, 16, 4, C.mb);
  // ベル
  const bx = w - 60, by = 27; X.save(); X.translate(bx, by);
  X.beginPath(); X.moveTo(-12, 8); X.quadraticCurveTo(-11, -12, 0, -13); X.quadraticCurveTo(11, -12, 12, 8); X.lineTo(15, 11); X.lineTo(-15, 11); X.closePath();
  X.strokeStyle = '#333'; X.lineWidth = 2.4; X.lineJoin = 'round'; X.stroke(); X.beginPath(); X.arc(0, 14, 4, 0, Math.PI); X.stroke(); X.restore();
  const arr = [.7, 1.3, 1.9]; let n = 0; arr.forEach(c => { if (t >= c) n++; });
  const clear = eio(seg(t, 3.0, 3.25));
  if (n > 0 && clear < 1) {
    const last = arr.filter(c => t >= c).pop(), pop = 1 - seg(t, last, last + .3);
    const s = (n === 1 ? eo(seg(t, .7, .9)) : 1) * (1 + .3 * pop) * (1 - clear);
    // 届いた合図の波紋
    const k = seg(t, last, last + .5); if (k > 0 && k < 1) ring(bx + 12, by - 12, 12 + 16 * k, `rgba(31,90,232,${.5 * (1 - k)})`, 2.5);
    X.save(); X.translate(bx + 12, by - 12); X.scale(s, s); circle(0, 0, 12, C.blue); ring(0, 0, 12, '#fff', 2.5); text(String(n), 0, 1, 14, '#fff', 800, 'center', NUMF); X.restore();
  }
  bar(24, 76, 220, C.line2, 12); bar(24, 102, 494); bar(24, 120, 400); fillRR(24, 140, 494, 40, 8, C.block);
  pointer([[.3, w + 30, h + 30], [2.3, w + 30, h + 30], [2.95, bx + 2, by + 4], [3.4, bx + 2, by + 4], [4.1, w + 30, h + 30]], [3.0], t);
});
