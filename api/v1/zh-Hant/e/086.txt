// 底線動畫 / Underline animation
// 類別包: lp (落地頁與網站效果) · element 14 · 分類: 滑鼠與點擊的互動
// 效果說明: Under the link the cursor is on, an orange line grows from the left; when the cursor leaves, the line shrinks away to the right.
// 動效規格:
//   - Three links (20 px, bold, 56 apart). The cursor rests on each, left to right, for about 1 s (hit area 6 px wider than the text on each side).
//   - Underline: orange, 3 thick, 8 px below the baseline.
//   - On hover, width 0→100% from the left end over 0.3 s (cubic ease-out).
//   - On leave, the left end moves right over 0.3 s, so the line exits to the right (like scaleX with transform-origin on the right).
// 適用情境: Use it on menu and text links to show where the pointer is, quietly.
// 標籤: underline, hover, link, navigation, menu, text-link, scalex
// 尺寸: 539x208 px · 週期: 4.8 s
// 依賴: kit lp -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 動態範例: https://budingsm.github.io/motion-prompt-zh/#086
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫方已清空畫布並完成 save/restore），w、h = 尺寸，u = [0, cycle) 內的秒數。畫面只由 u 決定。
// 以下是元素程式碼。CELLS、常數和輔助函式位於 kit 中。

// @element 14
// 14 下線アニメーション（リンクにカーソルを乗せると下線が左から伸び、離れると右へ抜ける）
CELLS[13] = (k, w, h, u) => {
  frame(k);
  const LINKS = ['サービス', '実績', '会社概要'], F = `800 20px ${JP}`, GAP = 56, LY = 84, DUR = 0.3;
  const cur = t => path([[0, 120, 150], [0.45, 170, 80], [1.3, 186, 84], [1.55, 268, 80], [2.4, 282, 84], [2.65, 372, 80], [3.5, 390, 84], [3.95, 430, 150], [4.8, 120, 150]], t);
  view(k, () => {
    logo(k, 22, 28); bars(k, 90, 132, 331, 2, C.faint, 6, 12, 0.6);
    k.font = F; const ws = LINKS.map(s => k.measureText(s).width); let x = (VW - ws.reduce((a, b) => a + b, 0) - GAP * (LINKS.length - 1)) / 2;
    LINKS.forEach((s, i) => {
      const x0 = x, x1 = x + ws[i], inside = t => { const [a, b] = cur(t); return a > x0 - 6 && a < x1 + 6 && b > LY - 22 && b < LY + 14; };
      txt(k, s, x0, LY, F, C.ink);
      // 最後に乗った・離れた時刻から 0.3秒（乗る: 左端から伸びる / 離れる: 左端から縮んで右へ抜ける）
      const on = inside(u); let tc = null; for (let j = 0, t = u; j < 600; j++, t -= 1 / 120) if (inside(t - 1 / 120) !== on) { tc = t; break; }
      if (tc !== null) {
        const q = eOut(P(u, tc, tc + DUR)), a = on ? 0 : q, b = on ? q : 1;
        if (b > a) { k.fillStyle = C.acc; k.fillRect(x0 + ws[i] * a, LY + 8, ws[i] * (b - a), 3); }
      }
      x += ws[i] + GAP;
    });
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
