// 頁面轉場 / Page transition
// 類別包: lp (落地頁與網站效果) · element 20 · 分類: 載入、轉場與持續動態
// 效果說明: Clicking a card in the work list sweeps an orange panel across from the left; the URL changes and the panel exits to the right on the detail page. "Back to list" returns the same way.
// 動效規格:
//   - Click on a card at 0.8 s. At 0.9–1.3 s a panel (orange, full viewport, logo in the centre) enters from the left and covers everything (cubic ease-in-out).
//   - At 1.3 s, while covered, the page and the URL in the address bar (/works → /works/02) switch.
//   - At 1.4–1.85 s the panel exits to the right (cubic ease-in-out).
//   - Click on "← Back to list" at 3.0 s: the panel enters at 3.1–3.5 s, the page switches at 3.5 s and the panel exits at 3.6–4.05 s.
//   - The panel covers the whole viewport, header included.
// 適用情境: Use it on brand sites and portfolios to link one page to the next.
// 標籤: page-transition, transition, view-transitions, navigation, wipe, barba, spa
// 尺寸: 539x208 px · 週期: 4.8 s
// 依賴: kit lp -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 動態範例: https://budingsm.github.io/motion-prompt-zh/#092
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫方已清空畫布並完成 save/restore），w、h = 尺寸，u = [0, cycle) 內的秒數。畫面只由 u 決定。
// 以下是元素程式碼。CELLS、常數和輔助函式位於 kit 中。

// @element 20
// 20 ページ遷移アニメーション（リンクを押すと、色の幕が横切って次のページに切り替わる）
CELLS[19] = (k, w, h, u) => {
  frame(k, u >= 1.3 && u < 3.5 ? 'example.com/works/02' : 'example.com/works');
  const CLK = [0.8, 3.0];
  const cover = t0 => eIO(P(u, t0, t0 + 0.4)), uncover = t0 => eIO(P(u, t0, t0 + 0.45));
  // 幕の左端: 入る −VW→0、出る 0→VW
  const xs = u < 1.4 ? -VW * (1 - cover(0.9)) : u < 3.1 ? VW * uncover(1.4) : u < 3.6 ? -VW * (1 - cover(3.1)) : VW * uncover(3.6);
  const pageB = u >= 1.3 && u < 3.5;
  const [cx, cy] = path([[0, 420, 150], [0.65, 255, 100], [2.6, 255, 100], [2.85, 46, 50], [3.3, 46, 50], [4.6, 420, 150], [4.8, 420, 150]], u);
  view(k, () => {
    header(k, 0);
    if (!pageB) {
      txt(k, 'WORKS', 34, 62, `800 18px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
      for (let i = 0; i < 3; i++) { const x = 34 + i * 152; pic(k, x, 74, 136, 60, i, 6); bars(k, x, 142, 100, 1, C.ink, 7); }
    } else {
      txt(k, '← 一覧へ', 34, 55, `700 12px ${JP}`, C.sub, 'left', 'middle');
      pic(k, 34, 68, 240, 90, 1, 8);
      txt(k, 'Project 02', 296, 96, `800 22px ${EN}`, C.ink); bars(k, 296, 110, 180, 3);
    }
    if (xs > -VW && xs < VW) { k.fillStyle = C.acc; k.fillRect(xs, 0, VW, VH); logo(k, xs + VW / 2 - 40, VH / 2, '#fff'); }
    click(k, cx, cy, u, CLK);
  });
};
