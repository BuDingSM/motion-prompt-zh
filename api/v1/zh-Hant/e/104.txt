// 藍調時刻 / Blue hour
// 類別包: photo (攝影風格、光線與鏡頭效果) · element 08 · 分類: 打光方式與時段
// 效果說明: Once the sun has set, the sky turns two bands of deep blue, the windows and street lamp come on in amber and the person becomes a dark blue shape.
// 動效規格:
//   - Scene 1: a head-and-shoulders figure in front of a horizon at y 150 (head centre x 270, y 78, radius 29 px, no face or hair). 
//   - Sun: radius 18 px at x 120, sinking from y 40 to y 190 while the effect reaches 70%, fading out between 40% and 80%
//   - Colour: sky #2C3B5E at the top and #4A6191 near the horizon, ground #283452, buildings #34446B, figure #1C2439
//   - Lights: one window in three and the lamp bulb turn #FFC75A; flat rings of 13 px and 8 px (#FFE3A3 at 35% and 60%) around the bulb
//   - Top-right time 12:00 → 18:35
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// 適用情境: For city and night scenes that need both a blue sky and glowing lights.
// 標籤: blue hour, twilight, dusk, night city, cityscape, deep blue sky
// 尺寸: 539x208 px · 週期: 5 s
// 依賴: kit photo -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 動態範例: https://budingsm.github.io/motion-prompt-zh/#104
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫方已清空畫布並完成 save/restore），w、h = 尺寸，u = [0, cycle) 內的秒數。畫面只由 u 決定。
// 以下是元素程式碼。CELLS、常數和輔助函式位於 kit 中。

// @element 08
// ブルーアワー: 日が沈んだ後、空が深い青になり、窓と街灯の暖かい光が灯る
CELLS[7] = cell((k, u) => {
  const a = env(u);
  const p = pal(BASE, NIGHT, a);
  stagePortrait(k, p, {
    sun: [120, lerp(40, 190, P(a, 0, 0.7)), 18, 1 - P(a, 0.4, 0.8)],
    after: D => { const k2 = D.k; k2.save(); k2.globalAlpha = 0.35 * a; D.circ(413, HZ - 53, 13, BASE.glow); k2.globalAlpha = 0.6 * a; D.circ(413, HZ - 53, 8, BASE.glow); k2.restore(); D.circ(413, HZ - 53, 4, p.bulb); }
  });
  bust(k, PX, PY, PR, p.person);
  hud(k, clock(lerp(720, 1115, a)));
});
