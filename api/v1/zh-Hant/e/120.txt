// 多重曝光 / Double exposure
// 類別包: photo (攝影風格、光線與鏡頭效果) · element 24 · 分類: 調色與風格
// 效果說明: The background turns white and the landscape of hills, house, trees and sun appears, scaled down, only inside the shape of the bust.
// 動效規格:
//   - Scene 3: a bust on the left (head centre x 140, y 82, radius 28 px) and hills, a house, trees and the sun (x 432, y 54, radius 15 px) on the right. 
//   - Background: white #FFFFFF outside the bust
//   - Inside the bust: the same landscape at 0.7×, placed so the sun sits at the top right of the head and the ridge of the hills runs across the neck
//   - Inside colours: sky #D3DBE6, hills #9AA8BD and #76869F, ground #55647E, trees #3F4C66
//   - Transition: the double exposure is laid over the original picture at the effect amount
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// 適用情境: For posters and covers that tie a person to a place, a memory or an inner world.
// 標籤: double exposure, multiple exposure, blend, silhouette, surreal, composite
// 尺寸: 539x208 px · 週期: 5 s
// 依賴: kit photo -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 動態範例: https://budingsm.github.io/motion-prompt-zh/#120
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫方已清空畫布並完成 save/restore），w、h = 尺寸，u = [0, cycle) 內的秒數。畫面只由 u 決定。
// 以下是元素程式碼。CELLS、常數和輔助函式位於 kit 中。

// @element 24
// 二重露光: 背景が白く飛び、胸像の形の中にだけ丘と家と木と太陽が写り込む
CELLS[23] = cell((k, u) => {
  const a = env(u);
  const DX = Object.assign({}, BASE, { sky: '#D3DBE6', sky2: '#D3DBE6', far: '#9AA8BD', mid: '#76869F', ground: '#55647E', bld: '#E8ECF2', edge: '#55647E', win: '#9AA8BD', winOff: '#9AA8BD', tree: '#3F4C66', trunk: '#3F4C66' });
  const draw = kk => {
    kk.fillStyle = '#FFFFFF'; kk.fillRect(0, 0, CW, CH);
    kk.save(); bustPath(kk, QX, QY, QR); kk.clip();
    // 風景を0.7倍にして、太陽が頭の右上、丘の稜線が首のあたりに来るように置く
    kk.translate(QX, QY); kk.scale(0.7, 0.7); kk.translate(-415, -77);
    kk.fillStyle = DX.sky; kk.fillRect(0, -200, CW * 2, CH * 3);
    stageLand(kk, DX, { person: false });
    kk.fillStyle = DX.ground; kk.fillRect(-CW, 200, CW * 3, CH * 3);
    kk.restore();
  };
  if (a >= 0.999) { draw(k); return; }   // 効果が入りきったら直接描く（拡大表示でも輪郭がくっきり）
  stageLand(k, BASE);
  if (a < 0.002) return;
  const [L, kl] = layer('dx'); draw(kl);   // 入り・戻りの途中は、別の面に描いて重ねる
  k.save(); k.globalAlpha = a; k.drawImage(L, 0, 0); k.restore();
});
