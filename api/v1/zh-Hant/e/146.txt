// 滾動文字顯現 / Scroll text reveal
// 類別包: lp2 (落地頁與網站效果（第二輯）) · element 02 · 分類: Scroll effects
// 效果說明: A light gray headline darkens one character at a time from the top left as the page scrolls, and fades back from the end when scrolling up.
// 動效規格:
//   - A 511 × 168 viewport. Scroll progress p: 0→1 over 0.3–2.6 s, 1→0 over 3.2–4.4 s (cubic ease-in-out). The page moves up by p × 40 px. One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - A three-line headline, 24 characters in all (27 px, weight 800, 38 px line height), all light gray (#D6DBE4) at the start.
//   - Character n (counting from 0) has darkness clamp(p × 26 − n): each character takes 1/26 of the progress to reach the text color (#1C2333), in step with the scroll.
//   - Scrolling back fades them from the last character by the same formula. The body-text bars and the scrollbar thumb move with the page.
// 適用情境: Use it on a short statement, such as a concept or a message from the founder, that should be read slowly.
// 標籤: scroll, text-reveal, scroll-driven, typography, opacity, storytelling
// 尺寸: 539x208 px · 週期: 9.4 s
// 依賴: kit lp2 -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/lp2.js
// 動態範例: https://budingsm.github.io/motion-prompt-zh/#146
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫方已清空畫布並完成 save/restore），w、h = 尺寸，u = [0, cycle) 內的秒數。畫面只由 u 決定。
// 以下是元素程式碼。CELLS、常數和輔助函式位於 kit 中。

// @element 02
// 02 スクロールテキストリビール（薄い見出しの文字が、スクロールに合わせて左から1文字ずつ濃くなる）
CELLS[1] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.6, 1], [3.2, 1], [4.4, 0]], u), y = p * 40;
  const LINES = ['毎日の小さな工夫が、', '暮らしを少しずつ', '変えていく。'], F = `800 27px ${JP}`;
  const total = LINES.join('').length;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'ABOUT US', 40, 34, `800 11px ${EN}`, C.acc, 'left', 'alphabetic', '2px');
    let n = 0;
    LINES.forEach((s, li) => {
      let x = 40; k.font = F;
      for (const ch of s) {
        const f = clamp(p * (total + 2) - n);
        txt(k, ch, x, 72 + li * 38, F, mix('#D6DBE4', C.ink, f));
        k.font = F; x += k.measureText(ch).width; n++;
      }
    });
    bars(k, 40, 168, 300, 3, C.faint, 6, 12, 0.7);
    k.restore();
    sbar(k, p, 0.55);
  });
};
