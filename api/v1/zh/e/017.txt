// 遮罩转场 / Mask transition
// 类别包: motion (动态图形动效) · element 17 · 分类: 转场与吸睛效果
// 效果说明: A circle- or star-shaped window grows, showing the next scene inside it.
// 动效规格:
//   - Day scene → night scene → day scene, 2.35 s each. The change runs at 0.45–1.6 s of each half (1.15 s), cubic ease-in-out.
//   - The next scene is drawn on top of the current one, clipped to a shape; the shape grows until it covers the frame.
//   - Day→night: a circle from the sun's centre (112, 104), radius 0→470.
//   - Night→day: a five-point star from the moon's centre (427, 104), outer radius 0→1150, rotating 1.2 rad (about 69°) as it grows.
//   - While it grows, the edge of the shape has a white outline (width 7).
// 适用场景: Use it to change scenes starting from an object in the frame, such as a sun, logo or button.
// 标签: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/motion.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#017
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
