// 形状转场 / Shape transition
// 类别包: motion (动态图形动效) · element 18 · 分类: 转场与吸睛效果
// 效果说明: Coloured bands sweep across and cover the frame; when they leave, the next scene is there.
// 动效规格:
//   - Day → night → day, 2.35 s each. Four horizontal bands (height 52; blue, coral, yellow, mint) whose ends are slanted by 80 px.
//   - Cover: 0.38–1.0 s. Each band's leading edge travels from one side of the frame to the other. Bands are delayed by 14% of the phase each, and each uses cubic ease-in-out over 58% of the phase.
//   - At 1.0 s, when everything is covered, the scene underneath is swapped for the next one.
//   - Uncover: 1.12–1.75 s. Each band's trailing edge follows with the same delays and curve, exposing the next scene.
//   - Direction: left to right for day→night, mirrored for night→day.
// 适用场景: Use it to change scenes with shapes in brand colours: chapter breaks, topic changes.
// 标签: shape-transition, transition, wipe, bands, stripes, cover, scene-change
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/motion.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#018
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 18
// 18 シェイプトランジション（色のついた帯が画面を覆い隠し、抜けると次の画面になっている）
CELLS[17] = abCell((k, w, h, u, a, b) => {
  const p1 = P(u, TA - 0.12, TA + 0.5), p2 = P(u, TA + 0.62, TA + 1.25), N = 4, sk = 80;
  SCR[p1 >= 1 ? b : a](k, w, h);
  const cols = [C.blue, C.coral, C.yel, C.mint];
  for (let i = 0; i < N; i++) {
    const d = i * 0.14, e1 = eIO(clamp((p1 - d) / (1 - 0.42))), e2 = eIO(clamp((p2 - d) / (1 - 0.42)));
    if (e1 <= 0 || e2 >= 1) continue;
    const bh = h / N + 1, y = i * h / N, span = w + sk + 20, dir = a === 0 ? 1 : -1;
    let xa = -sk - 10 + e2 * span, xb = -sk - 10 + e1 * span;
    k.save(); if (dir < 0) { k.translate(w, 0); k.scale(-1, 1); }
    k.fillStyle = cols[i]; k.beginPath(); k.moveTo(xa + sk, y); k.lineTo(xb + sk, y); k.lineTo(xb, y + bh); k.lineTo(xa, y + bh); k.closePath(); k.fill(); k.restore();
  }
});
