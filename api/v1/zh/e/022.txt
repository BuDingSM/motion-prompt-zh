// 扫光 / Light sweep
// 类别包: motion (动态图形动效) · element 22 · 分类: 转场与吸睛效果
// 效果说明: A band of light crosses the surface of a logo badge at an angle.
// 动效规格:
//   - One sweep per 2.35 s. The light is drawn only inside a clip of the badge (400 × 112, radius 28, blue gradient).
//   - Light band: a 116 px wide white gradient (opacity 0.72 in the middle, 0 at the edges) plus a thin 12 px band (opacity 0.5) 78 px away, rotated 0.42 rad (about 24°).
//   - It moves from 130 px left of the badge to 130 px right of it (660 px) at 0.45–1.5 s, quadratic ease-in-out.
//   - Sparkle: a four-point star at the top-right corner at 1.38–1.98 s; size 30 × sin(πq) × (1 + 0.2(1 − q)), rotating 0.9 rad.
//   - The badge itself does not move.
// 适用场景: Use it to give a logo, badge or button a premium finish.
// 标签: light-sweep, shine, sheen, glint, highlight, logo, badge, premium
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/motion.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#022
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 22
// 22 ライトスイープ（ロゴの表面を、光の帯が斜めに横切る）
CELLS[21] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, lu = mod(u, T), p = eIOs(P(lu, 0.45, 1.5));
  const BW = 400, BH = 112, X = w / 2 - BW / 2, Y = h / 2 - BH / 2;
  k.fillStyle = 'rgba(0,0,0,.3)'; rr(k, X + 3, Y + 9, BW, BH, 28); k.fill();
  const g = k.createLinearGradient(X, Y, X + BW, Y + BH); g.addColorStop(0, '#2B62E0'); g.addColorStop(1, '#4E8BFF'); k.fillStyle = g; rr(k, X, Y, BW, BH, 28); k.fill();
  k.fillStyle = C.yel; star(k, X + 64, Y + BH / 2, 30, 0, 0.5); k.lineJoin = 'round'; k.strokeStyle = C.yel; k.lineWidth = 5; k.stroke(); k.fill();
  txt(k, 'PREMIUM', X + 112, Y + BH / 2 + 3, `800 50px ${DISP}`, '#fff', 'left', 'middle', '3px');
  if (p > 0 && p < 1) {
    k.save(); rr(k, X, Y, BW, BH, 28); k.clip();
    k.translate(lerp(X - 130, X + BW + 130, p), Y + BH / 2); k.rotate(0.42);
    const gr = k.createLinearGradient(-58, 0, 58, 0); gr.addColorStop(0, 'rgba(255,255,255,0)'); gr.addColorStop(0.5, 'rgba(255,255,255,.72)'); gr.addColorStop(1, 'rgba(255,255,255,0)');
    k.fillStyle = gr; k.fillRect(-58, -160, 116, 320); k.fillStyle = 'rgba(255,255,255,.5)'; k.fillRect(78, -160, 12, 320);
    k.restore();
  }
  // 抜けぎわのきらめき
  const tt = lu - 1.38; if (tt > 0 && tt < 0.6) { const q = tt / 0.6, sc = Math.sin(Math.PI * q) * (1 + 0.2 * (1 - q)); k.save(); k.translate(X + BW - 8, Y + 6); k.rotate(q * 0.9); k.fillStyle = '#fff'; star(k, 0, 0, 30 * sc, 0, 0.22, 4); k.fill(); k.restore(); }
};
