// Toggle / Toggle switch
// 类别包: ui (UI 组件名称) · element 11 · 分类: 选择与切换元件
// 效果说明: The switches on the first two of three setting rows are turned on in turn, then off again.
// 动效规格:
//   - 4.5 s cycle on a 542×191 card. Three setting rows 60 px apart with dividers. At x 450 a 64×34 track (radius 17) and a white knob (radius 13, shadow).
//   - Clicks: row 1 on at 1.0 s, row 2 on at 1.8 s, row 1 off at 3.0 s, row 2 off at 3.6 s. Row 3 stays off.
//   - Each switch takes 0.2 s with easeOutCubic; the knob slides 30 px (17 → 47).
//   - The track colour blends from grey #d9d7d5 to blue #1f5ae8 with the switch progress.
// 适用场景: Use it for settings that take effect immediately and have only two states.
// 标签: toggle, switch, toggle-switch, on-off, boolean, settings
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/ui.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#059
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 11
// 11 トグルスイッチ（Toggle switch）
CELLS[10] = ui((w, h, t) => {
  const st = [eo(seg(t, 1.0, 1.2)) * (1 - eo(seg(t, 3.0, 3.2))), eo(seg(t, 1.8, 2.0)) * (1 - eo(seg(t, 3.6, 3.8))), 0];
  for (let i = 0; i < 3; i++) {
    const y = 14 + i * 60; bar(24, y + 10, [170, 140, 190][i], C.line2, 11); bar(24, y + 30, [260, 220, 240][i], C.line, 8);
    if (i < 2) { X.fillStyle = C.line; X.fillRect(24, y + 52, 494, 1.5); }
    const k = st[i], tx = 450, ty = y + 10;
    fillRR(tx, ty, 64, 34, 17, k > .5 ? C.blue : '#d9d7d5'); if (k > 0 && k < 1) { X.globalAlpha = k; fillRR(tx, ty, 64, 34, 17, C.blue); X.globalAlpha = 1; }
    shadowed(() => circle(tx + 17 + 30 * k, ty + 17, 13, '#fff'), 4, 1, .25);
  }
  pointer([[.3, w + 30, h + 30], [.95, 486, 30], [1.35, 486, 30], [1.75, 486, 90], [2.5, 486, 90], [2.95, 486, 30], [3.25, 486, 30], [3.55, 486, 90], [3.85, 486, 90], [4.3, w + 30, h + 30]], [1.0, 1.8, 3.0, 3.6], t);
});
