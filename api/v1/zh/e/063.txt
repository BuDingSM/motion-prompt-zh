// 数量加减器 / Stepper (+/−)
// 类别包: ui (UI 组件名称) · element 15 · 分类: 选择与切换元件
// 效果说明: Pressing + three times raises the number from 1 to 4, and pressing − three times brings it back to 1; the number pops on each press.
// 动效规格:
//   - 4.5 s cycle on a 542×191 card. A 200×60 box (radius 12) at (171, 52): "−" in the left 60 px, a blue "+" in the right 60 px, and a bold 30 px number in the middle.
//   - Clicks: + at 0.9, 1.35, 1.8 s (1 → 4), − at 2.8, 3.25, 3.7 s (4 → 1).
//   - The pressed segment is tinted light blue from 0.05 s before to 0.15 s after each click.
//   - Number: on each press its scale goes linearly from 1.18 back to 1 over 0.25 s, and it is blue meanwhile.
// 适用场景: Use it for small whole numbers changed one at a time, such as quantity or number of guests.
// 标签: stepper, quantity, number-input, increment, decrement, plus-minus, spin-button
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/ui.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#063
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 15
// 15 ステッパー（Stepper (+/−)）
CELLS[14] = ui((w, h, t) => {
  bar(171, 24, 90, C.line2, 10);
  const plus = [.9, 1.35, 1.8], minus = [2.8, 3.25, 3.7];
  let n = 1; plus.forEach(c => { if (t >= c) n++; }); minus.forEach(c => { if (t >= c) n--; });
  const last = [...plus, ...minus].filter(c => t >= c).pop();
  const pop = last !== undefined ? 1 - seg(t, last, last + .25) : 0;
  const bx = 171, by = 52, bw = 200, bh = 60;
  strokeRR(bx, by, bw, bh, 12, C.line2, 1.8);
  const pressP = plus.some(c => t >= c - .05 && t < c + .15), pressM = minus.some(c => t >= c - .05 && t < c + .15);
  if (pressM) fillRR(bx + 2, by + 2, 58, bh - 4, 10, C.lb); if (pressP) fillRR(bx + bw - 60, by + 2, 58, bh - 4, 10, C.lb);
  X.fillStyle = C.line2; X.fillRect(bx + 60, by + 10, 1.5, bh - 20); X.fillRect(bx + bw - 61, by + 10, 1.5, bh - 20);
  text('−', bx + 30, by + bh / 2, 30, '#555', 500, 'center', NUMF); text('+', bx + bw - 30, by + bh / 2, 30, C.blue, 500, 'center', NUMF);
  X.save(); X.translate(bx + bw / 2, by + bh / 2 + 1); X.scale(1 + .18 * pop, 1 + .18 * pop); text(String(n), 0, 0, 30, pop > .05 ? C.blue : C.ink, 700, 'center', NUMF); X.restore();
  bar(171, 136, 200, C.line); bar(171, 154, 140, C.line);
  pointer([[.3, w + 30, h + 30], [.85, 342, 86], [2.3, 342, 86], [2.75, 202, 86], [3.75, 202, 86], [4.3, w + 30, h + 30]], [...plus, ...minus], t);
});
