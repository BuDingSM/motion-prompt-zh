// 步骤指示器 / Step indicator
// 类别包: ui (UI 组件名称) · element 20 · 分类: 导览与状态元件
// 效果说明: Each click on "Next" turns the next of four step circles into a check and extends the blue line to the following step.
// 动效规格:
//   - 4.5 s cycle on a 542×191 card. Four steps ("Details, Review, Payment, Done" in Japanese) at x 80, 207, 334, 461, y 35; circles of radius 17 joined by a 4 px grey line.
//   - The cursor clicks "Next" (110×38, bottom right) at 0.9, 1.7 and 2.5 s.
//   - On each click the current step becomes a blue circle with a white check and the next becomes the current one (blue circle, white number). Steps not reached are white with a grey rim.
//   - The blue line grows 127 px per click over 0.3 s with easeInOutCubic; after the third click all four are checked.
//   - Reset: the indicator fades out over 3.75–4.05 s and fades back in at step 1 over 4.05–4.3 s.
// 适用场景: Use it in multi-step flows such as sign-up or checkout to show progress and what is left.
// 标签: step-indicator, stepper, steps, progress-steps, wizard, multi-step, checkout
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/ui.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#068
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 20
// 20 ステップインジケーター（Step indicator）
CELLS[19] = ui((w, h, t) => {
  const clicks = [.9, 1.7, 2.5]; let step = 0; clicks.forEach(c => { if (t >= c) step++; });
  const reset = seg(t, 3.75, 4.05), back = seg(t, 4.05, 4.3);
  const showStep = reset >= 1 ? 0 : step;
  const xs = [80, 207, 334, 461], labs = ['入力', '確認', '支払い', '完了'];
  const prog = clicks.reduce((s, c) => s + eio(seg(t, c, c + .3)), 0);
  X.save(); X.globalAlpha = reset < 1 ? 1 - reset : back;
  const pr = reset >= 1 ? 0 : prog;
  fillRR(xs[0], 33, xs[3] - xs[0], 4, 2, '#e3e1df'); fillRR(xs[0], 33, (xs[1] - xs[0]) * pr, 4, 2, C.blue);
  xs.forEach((x, i) => {
    const done = i < showStep || (showStep === 3 && i === 3), cur = i === showStep && showStep < 3;
    circle(x, 35, 17, done || cur ? C.blue : '#fff'); if (!(done || cur)) ring(x, 35, 16, C.line2, 2);
    if (done) check(x, 35, 7, '#fff', 2.6); else text(String(i + 1), x, 36, 16, cur ? '#fff' : '#9a9896', 700, 'center', NUMF);
    text(labs[i], x, 70, 14, done || cur ? C.blue : '#9a9896', 600, 'center');
  });
  X.restore();
  strokeRR(24, 96, 494, 30, 7, C.line2, 1.5); bar(36, 106, 160, C.line); bar(24, 148, 200, C.line);
  fillRR(w - 24 - 110, 142, 110, 38, 8, C.blue); text('次へ', w - 24 - 55, 161, 16, '#fff', 600, 'center');
  pointer([[.3, w + 30, h + 30], [.85, w - 74, 166], [2.6, w - 74, 166], [3.3, w + 30, h + 30]], clicks, t);
});
