// Sticky Header / Sticky header
// 类别包: lp (落地页与网站效果) · element 03 · 分类: 卷动时的效果
// 效果说明: The announcement strip scrolls away, but the header below it stops at the top edge and the content scrolls underneath.
// 动效规格:
//   - Scroll: 0→110 px at 0.4–1.3 s, 110→250 px at 1.9–2.8 s, 250→0 at 3.4–4.4 s (cubic ease-in-out).
//   - At the very top, a 22 px announcement strip (navy); below it a 34 px header (logo, three menu items, button).
//   - Header position = max(22 − scroll, 0): once the strip is gone, it stops at the top edge (position: sticky; top: 0).
//   - Over the next 24 px of scroll, a 12 px shadow under the header fades from 0 to 0.16 opacity.
//   - The content (photo, heading, three cards) scrolls normally under the header.
// 适用场景: Use it to keep the menu and the call-to-action within reach on a long page.
// 标签: sticky, sticky-header, fixed-header, position-sticky, navigation, header, scroll
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#075
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 03
// 03 追従ヘッダー（スクロールすると、ヘッダーが画面の上で止まってついてくる）
CELLS[2] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.4, 0], [1.3, 110], [1.9, 110], [2.8, 250], [3.4, 250], [4.4, 0]], u);
  const STRIP = 22;
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 70, 214, 112, 0, 8);
    txt(k, '新しい毎日を、', 262, 104, `800 22px ${JP}`, C.ink); bars(k, 262, 120, 210, 3); btn(k, 262, 160, 100, 26, '詳しく見る');
    k.fillStyle = C.soft; k.fillRect(0, 212, VW, 240);
    txt(k, '特長', VW / 2, 246, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 262, 136, 120, 8); k.fill(); pic(k, x + 8, 270, 120, 58, i, 5); bars(k, x + 10, 340, 110, 3, C.faint, 6, 11); }
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, STRIP);
    txt(k, '期間限定キャンペーン実施中', VW / 2, STRIP / 2 + 0.5, `600 11px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 250, 0.42);
  });
};
