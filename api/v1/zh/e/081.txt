// Hover 浮起效果 / Hover lift
// 类别包: lp (落地页与网站效果) · element 09 · 分类: 滑鼠与点击的互动
// 效果说明: Only the card under the cursor lifts slightly and its shadow spreads; it settles back when the cursor leaves.
// 动效规格:
//   - Three cards (146 × 128, radius 10). The cursor rests on each, left to right, for about 1 s.
//   - On hover the card rises 10 px over 0.3 s (cubic ease-out); on leave it returns over the same 0.3 s (like a CSS transition).
//   - Shadow, changed together with the lift: blur 6→22 px, y offset 2→12 px, opacity 0.08→0.22.
//   - Light grey page, white cards, so the shadow reads.
// 适用场景: Use it to make cards and buttons look clickable.
// 标签: hover, lift, float, card, shadow, translatey, interaction
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#081
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 09
// 09 ホバーで浮き上がる（カーソルが乗ったカードが、影をつけて少し持ち上がる）
CELLS[8] = (k, w, h, u) => {
  frame(k);
  const CX = [26, 192, 358], CY = 18, CWD = 146, CHT = 128, LIFT = 10;
  const cur = t => path([[0, 70, 160], [0.45, 96, 78], [1.4, 110, 86], [1.62, 262, 80], [2.5, 276, 88], [2.72, 428, 80], [3.6, 440, 88], [4.0, 470, 160], [4.8, 70, 160]], t);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    for (let i = 0; i < 3; i++) {
      const inside = t => { const [x, y] = cur(t); return x > CX[i] && x < CX[i] + CWD && y > CY && y < CY + CHT; };
      const q = trans(inside, u, 0.3), x = CX[i], y = CY - LIFT * q;
      k.save(); k.shadowColor = `rgba(28,35,51,${0.08 + 0.14 * q})`; k.shadowBlur = 6 + 16 * q; k.shadowOffsetY = 2 + 10 * q;
      k.fillStyle = '#fff'; rr(k, x, y, CWD, CHT, 10); k.fill(); k.restore();
      pic(k, x + 8, y + 8, CWD - 16, 66, i, 6);
      txt(k, ['山の宿', '海の宿', '星の宿'][i], x + 12, y + 96, `800 13px ${JP}`, C.ink); bars(k, x + 12, y + 106, 100, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
