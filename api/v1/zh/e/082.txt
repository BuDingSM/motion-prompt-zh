// Hover 图片放大 / Image zoom on hover
// 类别包: lp (落地页与网站效果) · element 10 · 分类: 滑鼠与点击的互动
// 效果说明: The photo under the cursor slowly enlarges while its frame stays the same size, and shrinks back when the cursor leaves.
// 动效规格:
//   - Two photo frames, 222 × 118 (radius 8). The cursor rests on the left one, then the right one, for about 1.5 s each.
//   - On hover the content scales 1→1.15 from the centre over 0.5 s (cubic ease-out); on leave it returns over 0.5 s.
//   - Anything outside the frame is clipped (overflow: hidden). The frame and the caption do not move.
// 适用场景: Use it on photo grids (work, products, articles) to show which item is about to be chosen.
// 标签: hover, zoom, image, scale, overflow-hidden, thumbnail, gallery
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#082
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 10
// 10 ホバーで画像ズーム（枠はそのままで、中の写真だけがゆっくり拡大する）
CELLS[9] = (k, w, h, u) => {
  frame(k);
  const FX = [22, 267], FY = 16, FW = 222, FH = 118, ZOOM = 0.15;
  const cur = t => path([[0, 60, 160], [0.5, 120, 70], [1.8, 140, 84], [2.2, 370, 70], [3.4, 390, 84], [3.9, 470, 160], [4.8, 60, 160]], t);
  view(k, () => {
    for (let i = 0; i < 2; i++) {
      const inside = t => { const [x, y] = cur(t); return x > FX[i] && x < FX[i] + FW && y > FY && y < FY + FH; };
      const q = trans(inside, u, 0.5);
      pic(k, FX[i], FY, FW, FH, i, 8, 1 + ZOOM * q);
      txt(k, ['夏の山小屋', '海辺のカフェ'][i], FX[i], FY + FH + 20, `800 13px ${JP}`, C.ink); bars(k, FX[i], FY + FH + 28, 150, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
