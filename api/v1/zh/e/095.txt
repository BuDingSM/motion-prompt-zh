// 渐层动画 / Animated gradient
// 类别包: lp (落地页与网站效果) · element 23 · 分类: 载入、转场与持续动态
// 效果说明: Behind the headline, soft patches of orange, light orange, blue and light blue drift in circles and keep blending differently.
// 动效规格:
//   - Base colour #FFF1EB, with four radial gradients on top (centre colour → transparent at the edge, each at 0.85 opacity).
//   - Orange (radius 230), light orange #FFB08F (210), blue (240), light blue #9DB6FB (200).
//   - Each centre travels an ellipse once per 4.8 s (x radii 70, 60, 80, 50; y radii 30, 40, 30, 35; start angles 0, 2.1, 4.2, 1.0 rad).
//   - A white headline (34 px) and a button sit on top; only the gradient moves.
// 适用场景: Use it to add movement and colour behind a hero or heading without using a photo.
// 标签: gradient, animated-gradient, mesh-gradient, background, hero, ambient, loop
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#095
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 23
// 23 グラデーションアニメーション（背景のグラデーションの色が、ゆっくり流れるように動き続ける）
CELLS[22] = (k, w, h, u) => {
  frame(k);
  const a = 2 * Math.PI * u / CYC;                        // 4.8秒で1周
  const BLOBS = [[C.acc, 150, 60, 230, 70, 30, 0], ['#FFB08F', 380, 40, 210, 60, 40, 2.1], [C.blue, 330, 150, 240, 80, 30, 4.2], ['#9DB6FB', 90, 170, 200, 50, 35, 1.0]];
  view(k, () => {
    k.fillStyle = '#FFF1EB'; k.fillRect(0, 0, VW, VH);
    for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
      const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
      const g = k.createRadialGradient(x, y, 0, x, y, r); g.addColorStop(0, col); g.addColorStop(1, 'rgba(255,255,255,0)');
      k.globalAlpha = 0.85; k.fillStyle = g; k.fillRect(0, 0, VW, VH); k.globalAlpha = 1;
    }
    txt(k, 'Hello, Future.', 40, 82, `800 34px ${EN}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 40, 100, 200, 2, 'rgba(255,255,255,.8)', 6, 12, 0.6);
    btn(k, 40, 128, 104, 26, 'はじめる', '#fff', C.ink);
  });
};
