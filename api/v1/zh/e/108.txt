// 追焦 / Panning shot
// 类别包: photo (摄影风格、光线与镜头效果) · element 12 · 分类: 镜头与相机效果
// 效果说明: The camera follows the moving car: the car stays sharp in the middle of the frame while only the buildings, trees and lamps behind it streak sideways.
// 动效规格:
//   - Street corner: buildings, trees and lamps behind a road at y 132–162 (no large person in front)
//   - Car: without the effect it drives from x −80 to 620 in 5 s; with it, it stays at x 270 ±10 px
//   - Streak: the background is drawn 12 times spread over 46 px sideways and averaged (spread 0 → 46 px)
//   - Wheels: a spoke line turns 12 times per 5 s
//   - Top-right shutter speed 1/1000 → 1/15
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// 适用场景: To show the speed of a car, bike or runner in a still photo.
// 标签: panning, panning shot, motion blur, speed, sports photography, slow shutter
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://budingsm.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 动态示例: https://budingsm.github.io/motion-prompt-zh/#108
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（调用方已清空画布并完成 save/restore），w、h = 尺寸，u = [0, cycle) 内的秒数。画面只由 u 决定。
// 以下是元素代码。CELLS、常量和辅助函数位于 kit 中。

// @element 12
// 流し撮り: 走る車を追ってカメラを振ると、車はくっきり、背景だけ横に流れる
CELLS[11] = cell((k, u) => {
  const a = env(u);
  const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
  const S = 46 * a, n = S < 0.5 ? 1 : 12;
  for (let i = 0; i < n; i++) {   // 背景を横にずらして重ね、平均をとる（流れ）
    k.save(); k.globalAlpha = 1 / (i + 1); k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0); stageStreet(k, BASE, { car: false, person: false }); k.restore();
  }
  const D = pen(k); car(D, x, BASE, { head: BASE.lit });
  // 車輪の回転（1周に12回転＝継ぎ目なし）
  k.save(); k.strokeStyle = BASE.tire; k.lineWidth = 1.6;
  for (const wx of [x - 21, x + 21]) { const t = u / CYC * TAU * 12; k.beginPath(); k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5); k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5); k.stroke(); }
  k.restore();
  hud(k, a > 0.5 ? '1/15' : '1/1000');
});
